cmake_minimum_required(VERSION 3.16)
project(measurement_hold)
find_package(ament_cmake REQUIRED)
foreach(dep controller_interface hardware_interface rclcpp rclcpp_lifecycle pluginlib sensor_msgs std_msgs robot_calibration)
  find_package(${dep} REQUIRED)
endforeach()
add_library(measurement_hold SHARED controller.cpp)
target_compile_features(measurement_hold PUBLIC cxx_std_17)
ament_target_dependencies(measurement_hold controller_interface hardware_interface rclcpp rclcpp_lifecycle pluginlib sensor_msgs std_msgs robot_calibration)
pluginlib_export_plugin_description_file(controller_interface plugin.xml)
install(TARGETS measurement_hold DESTINATION lib)
install(FILES hold.launch.py DESTINATION share/${PROJECT_NAME})
enable_testing()
add_executable(hold_test test.cpp)
target_compile_features(hold_test PUBLIC cxx_std_17)
add_test(NAME hold_core COMMAND hold_test)
ament_package()
